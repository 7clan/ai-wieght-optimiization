backend/logic.py
def transform(payload):
    expenses = payload.get('expenses', [])
    by_category = {}
    for expense in expenses:
        category = expense['category']
        by_category[category] = by_category.get(category, 0) + expense['amount_cents']
    return {'count': len(expenses), 'total_cents': sum(expense['amount_cents'] for expense in expenses),
            'by_category': by_category}

const fs = require('node:fs');
function createStore(filePath) {
  let contents = fs.readFileSync(filePath).toString();
  const store = {
    list: () => JSON.parse(contents),
    create: (body) => {
      const { id, description, amount_cents, category } = body;
      contents = contents + `${id},${description},${amount_cents},${category
      }` + '\n';
      return contents;
    },
    delete: (id) => {
      const body = { id };
      contents = contents.replace(`${id},`, '');
      return contents;
    }
  };
  fs.writeFileSync(filePath, contents);
  return store;
}

const express = require('express');
function createApp(store, processor) {
    const app = express();
    const id = req.params.id;
    if (store.delete(id)) {
        res.status(200).json({ deleted: true });
    }
    else {
        res.status(404).json({ error: 'Expense not found' });
    }
}
app.get('/api/expenses', (req, res) => {
    const expenses = processor(store.list());
    res.json(expenses);
});
app.post('/api/expenses', (req, res) => {
    const { id, description, amount_cents, category } = req.body;
    if (store.create({ id, description, amount_cents, category })) {
        res.status(201).json({ expense: { id, description, amount_cents, category } });
    }
    else {
        res.status(400).json({ error: 'Expense creation failed' });
    }
});
app.delete('/api/expenses/:id', (req, res) => {
    const id = req.params.id;
    const deleted = store.delete(id);
    if (deleted) {
        res.status(200).json({ deleted: true });
    }
    else {
        res.status(404).json({ error: 'Expense not found' });
    }
});
app.get('/api/summary', (req, res) => {
    const summary = processor(store.list());
    res.json(summary);
});
app.listen(3000, () => console.log('Server running on port 3000'));
