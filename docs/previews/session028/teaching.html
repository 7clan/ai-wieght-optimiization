  function create(body) {
    const list = list();
    const expenses = JSON.parse(list);
    const { url } = body;
    if (!url) return Error('Error: empty url');
    if (!Number.isInteger(parseInt(body.amount_cents))) return Error('Error: invalid amount_cents');
    const expense = { url, amount_cents: Number.parseInt(body.amount_cents) };
