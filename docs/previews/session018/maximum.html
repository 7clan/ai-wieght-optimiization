# Compute: maximum of payload.numbers, returning null (Python None) for an empty list.
# Input payload is a Python dict; return exactly a dict with key 'value'.
# value type: nullable-number.
def transform(payload):
  values = payload.get('numbers')
  if not values:
    return {'value': None}
  return {'value': max(values)}
