const fs = require('node:fs');
function createStore(filePath) {
  let entries = [];
  let id = 0;
  return {
    create: (body) => {
      if (!body.project || !body.minutes || !body.date) {
        throw new Error('Invalid entry');
      }
      if (body.project.length === 0 || body.minutes < 0 || body.date.length !== 10) {
        throw new Error('Invalid entry');
      }
      if (entries.some((entry) => entry.project === body.project)) {
        throw new Error('Invalid entry');
      }
      body.id = id++;
      entries.push(body);
      fs.writeFileSync(filePath, JSON.stringify(entries));
      return body;
    },
    list: () => {
      return entries;
    },
    delete: (id) => {
      if (!id) {
        throw new Error('Invalid id');
      }
      const index = entries.findIndex((entry) => entry.id === id);
      if (index === -1) {
        throw new Error('Invalid id');
      }
      entries.splice(index, 1);
      fs.writeFileSync(filePath, JSON.stringify(entries));
      return true;
    },
  };
}

const express = require('express');
function createApp(store, processor) {
    const app = express();
    app.use(express.json());
    app.get('/api/health', (req, res) => {
        res.status(200).json({ ok: true });
    });
    app.get('/api/entries', (req, res) => {
        res.status(200).json({ entries: store.list() });
    });
    app.post('/api/entries', (req, res) => {
        try {
            const entry = req.body;
            if (!entry.project || !entry.minutes || !entry.date || !entry.note) {
                res.status(400).json({ error: 'Invalid entry' });
            }
            else {
                const id = store.create(entry);
                res.status(201).json({ item: { id, ...entry } });
            }
        }
        catch (error) {
            res.status(400).json({ error: 'Invalid entry' });
        }
    });
    app.delete('/api/entries/:id', (req, res) => {
        try {
            const id = req.params.id;
            const deleted = store.delete(id);
            if (deleted) {
                res.status(200).json({ deleted });
            }
            else {
                res.status(404).json({ error: 'Entry not found' });
            }
        }
        catch (error) {
            res.status(400).json({ error: 'Invalid entry' });
        }
    });
    app.get('/api/projects', (req, res) => {
        const projects = Array.from(new Set(store.list()));
        projects.sort();
        res.status(200).json({ projects });
    });
    app.get('/api/summary', (req, res) => {
        const summary = processor({ entries: store.list() });
        res.status(200).json({ summary });
    });
    return app;
}
module.exports = createApp;
