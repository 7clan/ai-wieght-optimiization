const fs = require('node:fs');
function createStore(filePath) {
  let data = fs.readFileSync(filePath).toString();
  let entries = JSON.parse(data);
  return {
    list: function () {
      return entries;
    },
    create: function (body) {
      if (body.project.length === 0 || body.minutes === 0 || body.date.length === 0) {
        throw new Error('Invalid entry');
      }
      if (body.project.length < 1 || body.minutes < 1 || body.date.length < 1) {
        throw new Error('Invalid entry');
      }
      if (body.project === '' && body.minutes === 0 && body.date === '') {
        throw new Error('Invalid entry');
      }
      let id = entries.length.toString();
      if (entries[id] !== undefined) {
        throw new Error('Duplicate entry');
      }
      body.id = id;
      entries.push(body);
      data = JSON.stringify(entries);
      fs.writeFileSync(filePath, data);
      return body;
    },
    delete: function (id) {
      if (id.length < 1) {
        throw new Error('Invalid entry');
      }
      if (id.length > 1) {
        throw new Error('Invalid entry');
      }
      if (id === '') {
        throw new Error('Invalid entry');
      }
      if (entries[id] === undefined) {
        throw new Error('Invalid entry');
      }
      entries.splice(id, 1);
      data = JSON.stringify(entries);
      fs.writeFileSync(filePath, data);
      return true;
    },
  };
}

const express = require('express');
function createApp(store, processor) {
    const app = express();
    app.use(express.json());
    app.get('/api/health', function(req, res) {
        res.json({ok: true});
    });
    app.get('/api/entries', function(req, res) {
        return res.json({entries: store.list()});
    });
    app.post('/api/entries', function(req, res) {
        try {
            const entry = req.body;
            if (!entry.project) {
                throw new Error('Project is missing');
            }
            if (!entry.minutes) {
                throw new Error('Minutes is missing');
            }
            if (!entry.date) {
                throw new Error('Date is missing');
            }
            if (!entry.note) {
                throw new Error('Note is missing');
            }
            const project = store.create(entry);
            return res.json({item: project});
        } catch (e) {
            return res.status(400).json({error: e.message});
        }
    });
    app.delete('/api/entries/:id', function(req, res) {
        const id = req.params.id;
        const project = store.delete(id);
        if (!project) {
            return res.status(400).json({error: 'Invalid entry'});
        }
        return res.json({deleted: true});
    });
    app.get('/api/projects', function(req, res) {
        const projects = Array.from(new Set(store.list()));
        projects.sort();
        return res.json({projects});
    });
    app.get('/api/summary', function(req, res) {
        return res.json(processor({entries: store.list()}));
    });
    return app;
}
module.exports = createApp;
