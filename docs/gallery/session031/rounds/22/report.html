const fs = require('node:fs');
function createStore(filePath) {
    const entry = {};
    if (fs.existsSync(filePath)) {
        const json = fs.readFileSync(filePath).toString();
        entry = JSON.parse(json);
    }
    else {
        entry = [];
    }
    return {
        list: () => entry,
        create(body) {
            const newEntry = {
                id: '',
                project: body.project,
                minutes: body.minutes,
                date: body.date,
                note: body.note,
            };
            if (newEntry.project === '' || newEntry.minutes === 0) {
                throw new Error('Invalid entry');
            }
            newEntry.id = `${newEntry.project}-${newEntry.minutes}-${newEntry.date}`;
            entry.push(newEntry);
            fs.writeFileSync(filePath, JSON.stringify(entry));
            return true;
        },
        delete(id) {
            const index = entry.findIndex((entry) => entry.id === id);
            if (index !== -1) {
                entry.splice(index, 1);
                fs.writeFileSync(filePath, JSON.stringify(entry));
                return true;
            }
            return false;
        },
    };
}

const express = require('express');
function createApp(store, processor) {
    const app = express();
    app.use(express.json());
    app.get('/api/health', (req, res) => {
        res.json({ ok: true });
    });
    app.post('/api/entries', (req, res) => {
        const entry = req.body;
        if (!entry.project || !entry.minutes || !entry.date) {
            return res.status(400).json({ error: 'Invalid entry' });
        }
        const project = store.create(entry);
        res.status(201).json({ item: project });
    });
    app.delete('/api/entries/:id', (req, res) => {
        store.delete(req.params.id);
        res.json({ deleted: true });
    });
    app.get('/api/projects', (req, res) => {
        const projects = store.list().map((project) => {
            return { id: project.id, name: project.project };
        });
        res.json({ projects });
    });
    app.get('/api/summary', (req, res) => {
        res.json(processor({ entries: store.list() }));
    });
    return app;
}
