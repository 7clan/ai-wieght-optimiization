const fs = require('node:fs');
function createStore(filePath) {
  let entries = [];
  let id = 0;
  return {
    list: function () {
      return entries;
    },
    create: function (body) {
      if (!body.project || !body.minutes || !body.date) {
        throw new Error('Invalid entry');
      }
      if (entries.some((entry) => entry.project === body.project)) {
        throw new Error('Project already exists');
      }
      if (entries.some((entry) => entry.date === body.date)) {
        throw new Error('Date already exists');
      }
      if (entries.some((entry) => entry.minutes === body.minutes)) {
        throw new Error('Minutes already exists');
      }
      const entry = {
        id: id++,
        project: body.project,
        minutes: body.minutes,
        date: body.date,
        note: body.note,
      };
      entries.push(entry);
      fs.writeFileSync(filePath, JSON.stringify(entries));
      return entry;
    },
    delete: function (id) {
      const index = entries.findIndex((entry) => entry.id === id);
      if (index === -1) {
        return false;
      }
      entries.splice(index, 1);
      fs.writeFileSync(filePath, JSON.stringify(entries));
      return true;
    },
  };
}

const express = require('express');
function createApp(store, processor) {
  const app = express();
  app.use(express.json());
  app.get('/api/health', (req, res) => {
    res.json({ ok: true });
  });
  app.get('/api/entries', (req, res) => {
    res.json(store.list());
  });
  app.post('/api/entries', (req, res) => {
    try {
      const entry = processor(req.body);
      store.create(entry);
      res.json({ item: entry });
    } catch (err) {
      res.status(400).json({ error: 'Invalid entry' });
    }
  });
  app.delete('/api/entries/:id', (req, res) => {
    try {
      store.delete(req.params.id);
      res.json({ deleted: true });
    } catch (err) {
      res.status(400).json({ error: 'Invalid entry' });
    }
  });
  app.get('/api/projects', (req, res) => {
    res.json(store.list());
  });
  app.get('/api/summary', (req, res) => {
    res.json(processor({ entries: store.list() }));
  });
  return app;
}
module.exports = createApp;
